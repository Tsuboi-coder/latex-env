\section{GPTとは？}

\begin{frame}{ChatGPTって何？}

  もしも，あなたが小学生に
  「ChatGPTって何？」
  と質問されたら，どのように答えますか？

  \vspace{1em}

  \begin{itemize}
    \item 何でも質問できるAI？
    \item 文章を作ってくれるAI？
    \item ものすごく物知りなAI？
  \end{itemize}

  % [図案]
  % ChatGPTの入力欄と回答を簡略化したイラスト
  % 「質問 → ChatGPT → 回答」という非常に単純な構図でもよい

\end{frame}


\begin{frame}{学術的な正体}

  GPTは
  \textbf{Generative Pre-trained Transformer}
  の略である．

  \vspace{1em}

  \begin{itemize}
    \item \textbf{Generative}
      \begin{itemize}
        \item 新しい文章を生成する
      \end{itemize}

    \item \textbf{Pre-trained}
      \begin{itemize}
        \item 大量のデータを使って，あらかじめ学習する
      \end{itemize}

    \item \textbf{Transformer}
      \begin{itemize}
        \item 文章などの情報を扱うニューラルネットワーク
      \end{itemize}
  \end{itemize}

  % [図案]
  % GPT の G / P / T を横に並べ，
  % それぞれ Generative / Pre-trained / Transformer と対応させる

\end{frame}


\begin{frame}{生まれる疑問}

  では，GPTの正体を知ると，
  いくつか疑問が生まれる．

  \vspace{1em}

  \begin{itemize}
    \item なんで「何を質問しても知っている」のか

    \item なんで「テキストだけではなく画像なども扱える」のか

    \item なんで「間違えることがある」のか
  \end{itemize}

  \vspace{1em}

  この3つの疑問から，GPTの仕組みを見てみる．

  % [図案]
  % 中央に「GPT」
  % 周囲に「なぜ知ってる？」「なぜ画像も？」「なぜ間違う？」
  % という3つの吹き出し

\end{frame}

\section{GPTの作り方，動かし方}
% =========================================================

\begin{frame}{コンピュータは文章を読めない}

  コンピュータは，
  私たちと同じように文字を読んでいるわけではない．

  \vspace{1em}

  例えば，

  \begin{center}
    「今日は大学で勉強する」
  \end{center}

  という文章も，そのままでは計算できない．

  \vspace{1em}

  \begin{itemize}
    \item 文章を小さな単位に分割する
    \item それぞれを数字として表現する
    \item その数字を使って計算する
  \end{itemize}

  % [図案]
  % 「今日は大学で勉強する」
  %        ↓
  % 「今日」「は」「大学」「で」「勉強」「する」
  %        ↓
  % [数値] [数値] [数値] ...
  %
  % という3段階の模式図

\end{frame}


\begin{frame}{文章を「トークン」に分ける}

  GPTでは，文章を
  \textbf{トークン（token）}
  と呼ばれる単位に分けて扱う．

  \vspace{1em}

  例えば，

  \begin{center}
    「今日は大学で勉強する」
  \end{center}

  を，

  \begin{center}
    「今日」/「は」/「大学」/「で」/「勉強」/「する」
  \end{center}

  のような単位に分割する．

  \vspace{1em}

  \begin{itemize}
    \item 必ずしも「1単語 = 1トークン」ではない
    \item GPTはトークン単位で文章を処理する
  \end{itemize}

  % [図案]
  % 文章がハサミで分割されてトークンになるイラスト
  % 実際のTokenizerの例を使ってもよい

\end{frame}


\begin{frame}{ことばを数字にする}

  トークンを，コンピュータが計算できる
  \textbf{数字の並び}
  に変換する．

  \vspace{1em}

  \begin{itemize}
    \item 「猫」$\rightarrow$ 数字の並び
    \item 「犬」$\rightarrow$ 数字の並び
    \item 「自動車」$\rightarrow$ 数字の並び
  \end{itemize}

  \vspace{1em}

  このような数値表現を
  \textbf{埋め込み（Embedding）}
  と呼ぶ．

  \vspace{1em}

  意味や使われ方が似たものは，
  数値表現も似るように学習される．

  % [図案]
  % 2次元散布図風に
  % 「猫」「犬」を近く，
  % 「自動車」を離して配置
  %
  % 厳密なEmbeddingの可視化ではなく
  % 「意味が近いものは数字の世界でも近い」
  % というイメージ図でよい

\end{frame}


\begin{frame}{でも，単語だけでは意味が分からない}

  同じ言葉でも，
  周囲の文章によって意味が変わる．

  \vspace{1em}

  例えば，

  \begin{itemize}
    \item Appleを食べる
    \item Appleが新製品を発表する
  \end{itemize}

  \vspace{1em}

  「Apple」という文字だけでは，
  果物なのか会社なのか決められない．

  \vspace{1em}

  そこで必要になるのが，

  \begin{center}
    \textbf{「周りのどこを見ればよいか？」}
  \end{center}

  という考え方である．

  % [図案]
  % 「Apple」を中央に置き，
  % 左に fruit / eat / sweet
  % 右に company / iPhone / product
  % のように関連語を配置

\end{frame}


\begin{frame}{Attention：どこを見る？}

  Transformerの中心となる仕組みが
  \textbf{Attention}
  である．

  \vspace{1em}

  Attentionでは，

  \begin{center}
    「ある単語を考えるとき，
    他のどの単語をどれくらい見るか」
  \end{center}

  を計算する．

  \vspace{1em}

  例えば，

  \begin{center}
    太郎は花子に本を渡した．彼女は喜んだ．
  \end{center}

  という文章では，

  \begin{itemize}
    \item 「彼女」と「花子」の関係
    \item 「渡した」と「本」の関係
  \end{itemize}

  などを考えながら文章を処理する．

  % [図案]
  % 文中の各単語を横に並べ，
  % 「彼女」→「花子」に太い線
  % 「渡した」→「本」に線
  % 関係の強さによって線の太さを変える

\end{frame}


\begin{frame}{Transformerが文脈を作る}

  Attentionによって，
  単語は周囲の情報を取り込む．

  \vspace{1em}

  \begin{itemize}
    \item 単語そのものの情報
    \item 周囲の単語との関係
    \item 文章全体の文脈
  \end{itemize}

  を使って，
  各トークンの表現を更新していく．

  \vspace{1em}

  Transformerでは，
  このような処理を何層も繰り返す．

  \vspace{1em}

  \begin{center}
    \textbf{文章を読みながら，
    文脈に応じて表現を作り直していく}
  \end{center}

  % [図案]
  % Token
  %   ↓
  % Transformer
  %   ↓
  % Transformer
  %   ↓
  % Transformer
  %   ↓
  % 文脈を含んだ表現
  %
  % という積み重ねの図

\end{frame}


\begin{frame}{GPTが学んでいること}

  GPTの基本的な学習は，
  実は非常にシンプルである．

  \vspace{1em}

  \begin{center}
    \textbf{「次に来るトークンを当てる」}
  \end{center}

  \vspace{1em}

  例えば，

  \begin{center}
    日本の首都は \underline{\hspace{3em}}
  \end{center}

  に対して，

  \begin{itemize}
    \item 東京：高い確率
    \item 京都：低い確率
    \item バナナ：非常に低い確率
  \end{itemize}

  となるように学習する．

  % [図案]
  % 「日本の首都は」
  %        ↓
  % 東京  92%
  % 京都   3%
  % 大阪   2%
  % ...
  %
  % のような棒グラフ

\end{frame}


\begin{frame}{「次を当てる」を繰り返す}

  文章を生成するときも，
  基本は同じである．

  \vspace{1em}

  \begin{enumerate}
    \item これまでの文章を読む
    \item 次のトークンの確率を計算する
    \item 次のトークンを選ぶ
    \item そのトークンを文章に追加する
    \item また次を予測する
  \end{enumerate}

  \vspace{1em}

  これを何度も繰り返すことで，
  長い文章を生成する．

  % [図案]
  % 「吾輩は」
  %    ↓
  % 「吾輩は猫」
  %    ↓
  % 「吾輩は猫で」
  %    ↓
  % 「吾輩は猫である」
  %
  % と1トークンずつ伸びるアニメーション向き

\end{frame}


\begin{frame}{なぜ「何でも知っている」ように見える？}

  GPTは，大量の文章を使って
  「次を当てる」学習を繰り返す．

  \vspace{1em}

  正確に予測するためには，

  \begin{itemize}
    \item 単語の使い方
    \item 文法
    \item 文章の構造
    \item 世の中について書かれた情報
    \item 質問と回答のパターン
    \item プログラムの書き方
  \end{itemize}

  など，さまざまな規則性を利用する必要がある．

  \vspace{1em}

  その結果，

  \begin{center}
    \textbf{多くのことを知っているように振る舞える}
  \end{center}

  ようになる．

  % [図案]
  % 大量の本・Webページ・コード
  %       ↓
  % 「次のトークン予測」
  %       ↓
  % GPT
  %
  % という学習イメージ

\end{frame}

\section{GPTには目と耳がある？}
% =========================================================

\begin{frame}{GPTには目と耳がある？}

  最近のChatGPTは，
  テキスト以外も扱うことができる．

  \vspace{1em}

  \begin{itemize}
    \item 画像
    \item 音声
    \item 動画
  \end{itemize}

  \vspace{1em}

  しかし，GPTがそのまま
  「目」や「耳」を持っているわけではない．

  \vspace{1em}

  ポイントは，

  \begin{center}
    \textbf{異なる情報を，
    Transformerが扱える表現に変換する}
  \end{center}

  ことである．

  % [図案]
  % テキスト → Text Encoder
  % 画像     → Vision Encoder
  % 音声     → Audio Encoder
  %
  %             ↓
  %
  %         Transformer
  %
  % という模式図

\end{frame}


\begin{frame}{画像も「数字」にする}

  テキストでは，
  トークンを数字に変換した．

  \vspace{1em}

  画像でも基本的な考え方は同じである．

  \begin{itemize}
    \item 画像を小さな領域などに分ける
    \item 画像の特徴を数字として表現する
    \item Transformerが扱える形に変換する
  \end{itemize}

  \vspace{1em}

  つまり，

  \begin{center}
    \textbf{テキストも画像も，
    最終的には「数字の表現」として扱われる}
  \end{center}

  % [図案]
  % 左：
  % 「猫が寝ている」→ Token → 数値表現
  %
  % 右：
  % 猫の画像 → Image Patch / Encoder → 数値表現
  %
  % 最後に両方がTransformerへ入る図

\end{frame}


\begin{frame}{マルチモーダルモデル}

  テキスト，画像，音声など，
  複数種類の情報を扱うモデルを

  \begin{center}
    \textbf{マルチモーダルモデル}
  \end{center}

  と呼ぶ．

  \vspace{1em}

  \begin{itemize}
    \item 画像を見て説明する
    \item 写真について質問に答える
    \item 音声を聞いて文章にする
    \item テキストと画像を組み合わせて推論する
  \end{itemize}

  \vspace{1em}

  Transformerは，
  「文章だけを処理する仕組み」というより，

  \begin{center}
    \textbf{さまざまな情報の関係を扱える仕組み}
  \end{center}

  と考えることができる．

  % [図案]
  % Image / Text / Audio / Video
  %        ↓
  % 共通の表現空間
  %        ↓
  % Transformer
  %        ↓
  % Text / Voice / etc.

\end{frame}

\section{それでも間違う！}
% =========================================================

\begin{frame}{それでも間違う！}

  GPTは非常に多くのことを答えられる．

  \vspace{1em}

  しかし，

  \begin{itemize}
    \item 存在しない論文を紹介する
    \item 間違った歴史的事実を説明する
    \item 計算を間違える
    \item 自信満々に誤った説明をする
  \end{itemize}

  ことがある．

  \vspace{1em}

  このように，
  もっともらしい誤った内容を生成する現象を

  \begin{center}
    \textbf{ハルシネーション}
  \end{center}

  と呼ぶ．

  % [図案]
  % 左：ユーザー「○○という論文を教えて」
  % 右：GPTが架空の論文タイトルを堂々と回答
  %
  % 実例よりも架空例の方が安全で分かりやすい

\end{frame}


\begin{frame}{なぜ間違える？}

  GPTの目的は，

  \begin{center}
    「世界について正しい文章を書くこと」
  \end{center}

  ではなく，

  \begin{center}
    \textbf{「文脈から次に来そうなトークンを予測すること」}
  \end{center}

  \vspace{1em}

  そのため，

  \begin{itemize}
    \item 文章として自然
    \item 文法的にも正しい
    \item 一見もっともらしい
  \end{itemize}

  だけど

  \begin{center}
    \textbf{事実として正しいとは限らない}という問題がある．
  \end{center}

  % [図案]
  % 「もっともらしさ」と「正しさ」の2軸
  %
  % もっともらしい ＆ 正しい
  % もっともらしい ＆ 間違い ← Hallucination
  %
  % という2×2図

\end{frame}


\begin{frame}{間違いを減らすには？}

  ハルシネーションを完全になくすことは難しいが，
  減らす方法はある．

  \vspace{1em}

  \begin{columns}

    \begin{column}{0.48\textwidth}
      \textbf{使い手側}

      \begin{itemize}
        \item 出典を確認する
        \item 重要な情報は裏取りする
        \item 条件を明確に伝える
        \item AIの回答を鵜呑みにしない
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{システム側}

      \begin{itemize}
        \item 検索結果を参照させる
        \item データベースを利用する
        \item 外部ツールで計算する
        \item 回答を検証する仕組みを入れる
      \end{itemize}
    \end{column}

  \end{columns}

  % [図案]
  % 左：User
  % 右：AI System
  % と分けて対策を配置
  %
  % RAGについて触れるなら
  % 「質問 → 検索 → 情報 → GPT → 回答」
  % の簡単な図を置く

\end{frame}


\begin{frame}{3つの疑問を振り返る}

  最初の3つの疑問に戻ってみる．

  \vspace{1em}

  \begin{itemize}
    \item \textbf{なんで「何を質問しても知っている」のか}
      \begin{itemize}
        \item 大量の文章から，
              次のトークンを予測する学習をしているから
      \end{itemize}

    \item \textbf{なんで画像なども扱えるのか}
      \begin{itemize}
        \item 画像なども数値表現に変換し，
              Transformerで扱えるから
      \end{itemize}

    \item \textbf{なんで間違えることがあるのか}
      \begin{itemize}
        \item 「正しい答え」ではなく，
              「もっともらしい次」を予測しているから
      \end{itemize}
  \end{itemize}

  % [図案]
  % 冒頭の「3つの疑問」のスライドと
  % 全く同じレイアウトを再利用し，
  % 今度は各疑問の下に答えを表示する
  %
  % 冒頭と終盤が対応するので非常に見栄えがよい

\end{frame}


\begin{frame}{ChatGPTの "T" って？}

  \begin{center}

    \Large
    Transformerは，

    \vspace{1em}

    \textbf{情報同士の関係を見ながら，\\
    文脈に応じた表現を作る仕組み}

    \vspace{1.5em}

    \normalsize
    そしてGPTは，

    \vspace{0.5em}

    \textbf{そのTransformerを使って\\
    「次に何が来るか」を学んだモデル}

  \end{center}

\end{frame}
